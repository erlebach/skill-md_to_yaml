1. Create schema/parser.py:
   ```python
   """Hybrid YAML+Markdown deck file parser.

   File format:
   - File starts with --- and ends blocks with ---
   - First block between --- markers is deck metadata (YAML)
   - Subsequent blocks are slides, each with:
     - YAML frontmatter between --- markers
     - Optional Markdown body after closing ---
   - Slides are separated by the next --- (which opens the next slide's frontmatter)

   Parsing strategy:
   1. Split file into blocks using regex ^---$ at start of line
   2. First non-empty block is deck metadata
   3. Remaining blocks alternate: odd = frontmatter YAML, even = Markdown body
      (because --- closes frontmatter and body runs until next ---)
   """
   import re
   from pathlib import Path
   import yaml
   from pydantic import TypeAdapter, ValidationError
   from schema.models import DeckMetadata, AnySlide, Deck

   FENCE_RE = re.compile(r'^---\s*$', re.MULTILINE)

   _slide_adapter = TypeAdapter(AnySlide)

   def parse_deck_file(path: str | Path) -> Deck:
       text = Path(path).read_text(encoding='utf-8').strip()

       # Split on --- lines
       parts = FENCE_RE.split(text)
       # Filter out empty parts
       parts = [p for p in parts if p.strip()]

       if not parts:
           raise ValueError("Empty deck file")

       # First part is deck metadata
       metadata_raw = yaml.safe_load(parts[0])
       if not isinstance(metadata_raw, dict):
           raise ValueError(f"Deck metadata must be a YAML mapping, got {type(metadata_raw).__name__}")
       metadata = DeckMetadata(**metadata_raw)

       # Remaining parts: pair up frontmatter and body
       # Each slide is: frontmatter_yaml [optional body]
       # After splitting on ---, slides appear as:
       #   parts[1] = slide1 frontmatter
       #   parts[2] = slide1 body + possibly empty (if no body before next ---)
       # BUT this depends on exact format. The simpler approach:
       # Re-parse the file by finding slide blocks.

       slides = []
       # Alternative: re-split more carefully
       # After metadata, reassemble remaining text and split into slide blocks
       # Each slide block = frontmatter\n---\nbody (body until next --- or EOF)

       remaining = text
       # Skip past the metadata block (first --- ... ---)
       # Find the closing --- of metadata
       match_positions = list(FENCE_RE.finditer(remaining))
       if len(match_positions) < 2:
           return Deck(metadata=metadata, slides=[])

       # After metadata closing ---, the rest is slides
       after_metadata = remaining[match_positions[1].end():]

       # Each slide starts with --- (frontmatter open)
       # Split into slide chunks: each chunk is frontmatter\n---\nbody
       slide_chunks = _split_into_slides(after_metadata)

       for frontmatter_str, body_str in slide_chunks:
           frontmatter = yaml.safe_load(frontmatter_str)
           if not isinstance(frontmatter, dict):
               continue
           frontmatter['body'] = body_str.strip() if body_str else None
           slide = _slide_adapter.validate_python(frontmatter)
           slides.append(slide)

       return Deck(metadata=metadata, slides=slides)


   def _split_into_slides(text: str) -> list[tuple[str, str]]:
       """Split remaining text into (frontmatter, body) pairs.

       Each slide is:
         ---
         frontmatter YAML
         ---
         optional body text
       """
       slides = []
       fence_matches = list(FENCE_RE.finditer(text))

       i = 0
       while i < len(fence_matches):
           # Opening ---
           fm_start = fence_matches[i].end()
           # Find closing ---
           if i + 1 < len(fence_matches):
               fm_end = fence_matches[i + 1].start()
               frontmatter = text[fm_start:fm_end]
               # Body is everything after closing --- until next opening ---
               body_start = fence_matches[i + 1].end()
               if i + 2 < len(fence_matches):
                   body_end = fence_matches[i + 2].start()
                   body = text[body_start:body_end]
               else:
                   body = text[body_start:]
               slides.append((frontmatter, body))
               i += 2  # Move past this slide's two --- markers
           else:
               # Only opening ---, treat rest as frontmatter with no body
               frontmatter = text[fm_start:]
               slides.append((frontmatter, ''))
               break

       return slides
   ```
