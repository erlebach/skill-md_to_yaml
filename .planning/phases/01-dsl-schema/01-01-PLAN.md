1. Create schema/json_schema.py:
   ```python
   import json
   from pathlib import Path
   from schema.models import Deck

   def generate_schema() -> dict:
       """Generate JSON Schema from Deck Pydantic model."""
       return Deck.model_json_schema()

   def write_schema(output_path: str | Path = None) -> Path:
       """Write JSON Schema to file. Default: schema/deck.schema.json"""
       if output_path is None:
           output_path = Path(__file__).parent / 'deck.schema.json'
       output_path = Path(output_path)
       schema = generate_schema()
       output_path.write_text(json.dumps(schema, indent=2) + '\n')
       return output_path

   if __name__ == '__main__':
       path = write_schema()
       print(f'Schema written to {path}')
   ```

2. Run `python -m schema.json_schema` to generate schema/deck.schema.json
